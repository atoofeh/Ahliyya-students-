`client/src/hooks/useFocusTrap.js`:
```js
import { useEffect } from 'react';

const FOCUSABLE = 'a[href],button:not([disabled]),textarea:not([disabled]),input:not([disabled]),select:not([disabled]),[tabindex]:not([tabindex="-1"])';

/**
 * Keeps keyboard focus inside `ref` while `active`, closes on Escape, and
 * returns focus to whatever was focused before the trap opened.
 */
export function useFocusTrap(ref, { active, onEscape }) {
  useEffect(() => {
    if (!active) return undefined;
    const opener = document.activeElement;
    const node = ref.current;
    const first = node?.querySelector(FOCUSABLE);
    (first || node)?.focus?.();

    const onKeyDown = (e) => {
      if (e.key === 'Escape') { e.stopPropagation(); onEscape?.(); return; }
      if (e.key !== 'Tab' || !node) return;
      const items = [...node.querySelectorAll(FOCUSABLE)].filter((el) => el.offsetParent !== null);
      if (items.length === 0) return;
      const firstEl = items[0];
      const lastEl = items[items.length - 1];
      if (e.shiftKey && document.activeElement === firstEl) { e.preventDefault(); lastEl.focus(); }
      else if (!e.shiftKey && document.activeElement === lastEl) { e.preventDefault(); firstEl.focus(); }
    };
    document.addEventListener('keydown', onKeyDown, true);
    return () => {
      document.removeEventListener('keydown', onKeyDown, true);
      if (opener instanceof HTMLElement) opener.focus();
    };
  }, [ref, active, onEscape]);
}
```

`client/src/hooks/useNetworkStatus.js`:
```js
import { useEffect, useState } from 'react';

/** Browser connectivity. Not a server reachability check — just navigator.onLine + events. */
export function useNetworkStatus() {
  const [online, setOnline] = useState(() => (typeof navigator === 'undefined' ? true : navigator.onLine));
  useEffect(() => {
    const up = () => setOnline(true);
    const down = () => setOnline(false);
    window.addEventListener('online', up);
    window.addEventListener('offline', down);
    return () => { window.removeEventListener('online', up); window.removeEventListener('offline', down); };
  }, []);
  return { online };
}
```

```jsx
 * Accessible quiz choices. Single-answer questions are a radiogroup with roving
 * focus and arrow-key selection; multi-answer questions are checkboxes.
 */
export default function ChoiceGroup({ multiple, options, value = [], onChange, name }) {
  const refs = useRef([]);
  const selected = (id) => value.includes(id);

  const toggle = (id) => {
    if (multiple) onChange(selected(id) ? value.filter((v) => v !== id) : [...value, id]);
    else onChange([id]);
  };

  const onKeyDown = (e, i) => {
    const last = options.length - 1;
    let next = null;
    if (e.key === 'ArrowDown' || e.key === 'ArrowRight') next = i === last ? 0 : i + 1;
    if (e.key === 'ArrowUp' || e.key === 'ArrowLeft') next = i === 0 ? last : i - 1;
    if (next != null) {
      e.preventDefault();
      refs.current[next]?.focus();
      if (!multiple) toggle(options[next].id);
      return;
    }
    if (e.key === ' ' || e.key === 'Enter') { e.preventDefault(); toggle(options[i].id); }
  };

  const focusIndex = Math.max(0, options.findIndex((o) => selected(o.id)));

  return (
    <div role={multiple ? 'group' : 'radiogroup'} aria-label={name} className="space-y-2">
      {options.map((o, i) => {
        const on = selected(o.id);
        return (
          <button
            key={o.id}
            type="button"
            ref={(el) => { refs.current[i] = el; }}
            role={multiple ? 'checkbox' : 'radio'}
            aria-checked={on}
            tabIndex={multiple ? 0 : i === focusIndex ? 0 : -1}
            onClick={() => toggle(o.id)}
            onKeyDown={(e) => onKeyDown(e, i)}
```

```jsx
        const { data } = await api.post(`/quizzes/attempts/${attempt.attemptId}/answers`, { answers: payload, clientSavedAt: new Date().toISOString() });
        if (data.saved) { setSavedAt(new Date()); }
        setSaveState('saved');
      } catch { setSaveState('offline'); }
    };
    const interval = setInterval(save, 15000);
    window.addEventListener('online', save);
    return () => { clearInterval(interval); window.removeEventListener('online', save); };
```

```jsx
import { useEffect, useState, useCallback } from 'react';
import { MessageSquareText, Send } from 'lucide-react';
import api, { apiError } from '../../../api/axios';
import { useToast } from '../../../hooks/useToast';
import { Card, Button, Input, Badge, EmptyState } from '../../../components/ui';
import { SkeletonCard } from '../../../components/Skeletons';
import ErrorState from '../../../components/ErrorState';

/** Course Q&A for staff: answer a thread and it is marked resolved. */
export default function QaTab({ courseId }) {
  const toast = useToast();
  const [threads, setThreads] = useState(null);
  const [error, setError] = useState(null);
  const [reply, setReply] = useState({});

  const load = useCallback(async () => {
    setError(null);
    try {
      const { data } = await api.get(`/qa/courses/${courseId}`);
      setThreads(data.items);
    } catch (err) { setError({ code: err.response?.data?.code, message: apiError(err) }); }
  }, [courseId]);
  useEffect(() => { load(); }, [load]);

  const sendReply = async (threadId) => {
    const body = reply[threadId]?.trim();
    if (!body) return;
    try {
      await api.post(`/qa/${threadId}/replies`, { body });
      setReply((r) => ({ ...r, [threadId]: '' }));
      toast.success('Reply posted — the thread is marked resolved');
      load();
    } catch (err) { toast.error(apiError(err)); }
  };

  if (error) return <ErrorState code={error.code} message={error.message} onRetry={load} />;
  if (!threads) return <div className="space-y-3"><SkeletonCard /><SkeletonCard /></div>;
  if (threads.length === 0) {
    return <Card><EmptyState icon={<MessageSquareText size={24} />} title="No student questions yet"
      subtitle="Questions students ask in this course appear here." /></Card>;
  }

  return (
    <div className="space-y-3">
      {threads.map((t) => (
        <Card key={t._id} className="p-4">
          <div className="flex items-center justify-between gap-2">
            <p className="font-medium text-surface-900 dark:text-white">{t.title}</p>
            <Badge color={t.resolved ? 'green' : 'amber'}>{t.resolved ? 'Resolved' : 'Open'}</Badge>
          </div>
          <p className="mt-1 text-sm text-surface-600 dark:text-surface-300">{t.body}</p>
          <p className="mt-1 text-xs text-surface-400">{t.author?.name} · {new Date(t.createdAt).toLocaleString()}</p>
          {t.replies.map((r) => (
            <div key={r._id} className="mt-2 rounded-lg bg-surface-50 p-2.5 text-sm dark:bg-surface-700/40">
              <p className="text-surface-700 dark:text-surface-200">{r.body}</p>
              <p className="mt-0.5 text-xs text-surface-400">{r.author?.name}</p>
            </div>
          ))}
          <div className="mt-2 flex gap-2">
            <Input placeholder="Reply as instructor (marks the thread resolved)…" value={reply[t._id] || ''}
              onChange={(e) => setReply((r) => ({ ...r, [t._id]: e.target.value }))} />
            <Button size="sm" onClick={() => sendReply(t._id)} aria-label="Send reply"><Send size={14} /></Button>
          </div>
        </Card>
      ))}
    </div>
  );
}
```
